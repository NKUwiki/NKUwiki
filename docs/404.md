<script setup>
import { onMounted } from 'vue'

// cleanUrls 下站内地址不带尾斜杠（/map 而不是 /map/），但历史外链、搜索引擎
// 已收录的地址与手敲的网址常常带斜杠，GitHub Pages 会直接落到这个 404 页。
// 只在「确实 404 了」的前提下把尾斜杠去掉重试一次：目录页（/categories/）本身
// 返回 200，不会执行到这里，所以不存在互相跳转的死循环。
onMounted(() => {
  const { pathname, search, hash } = window.location
  if (pathname.length > 1 && pathname.endsWith('/'))
    window.location.replace(pathname.slice(0, -1) + search + hash)
})
</script>
